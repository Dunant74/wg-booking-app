public static User findUser(Map<String, User> users, String email) throws UserNotFoundExecption {
        if(!users.containsKey(email)){
            throw new UserNotFoundExecption("User not found : " + email);
        }
        return users.get(email);
}

8. Lambda und Streams

public static void main(String[] args) {

            List<User> users = new ArrayList<>();

            users.add( new  User("Alice", "alice@example.com"));
            users.add(new User("Bob", "bob@example.com"));
            users.add(new User("Anna", "anna@example.com"));

            System.out.println("--- ALL users -- ");
            users.forEach(u -> u.displayInfo());

            System.out.println("\n---User starting with A -- ");
            users.stream()
                    .filter(u -> u.getName().startsWith("A"))
                    .forEach(u -> u.displayInfo());

            System.out.println("\n--- Emails ----");
            users.stream()
                    .map(u -> u.getEmail())
                    .forEach(email -> System.out.println(email));


            long count  = users.stream()
                    .filter(u -> u.getName().startsWith("A"))
                    .count();

            System.out.println("\nCount: " + count);

        }
